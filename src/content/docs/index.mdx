import { Card, CardGrid } from '@astrojs/starlight/components';

<script>{`
// Platform-aware download: point the primary hero button at the direct installer for
// this OS, using the same latest.json the app's updater reads (served at the site root).
// If anything here fails, the button keeps its releases-page fallback link.
(async () => {
  try {
    const btn = document.querySelector('.hero a[href$="/releases/latest"]');
    if (!btn) return;
    const isWin = /Windows/i.test(navigator.userAgent);
    const isMac = /Macintosh|Mac OS X/i.test(navigator.userAgent);
    if (!isWin && !isMac) return; // Linux et al: keep the releases-page link
    const res = await fetch('/atrium-site/latest.json');
    if (!res.ok) return;
    const { version } = await res.json();
    if (!/^[0-9A-Za-z.-]+$/.test(version)) return;
    const asset = isWin
      ? 'Atrium_' + version + '_x64-setup.exe'
      : 'Atrium_' + version + '_universal.dmg';
    btn.href = 'https://github.com/redlinelabs-dev/atrium-site/releases/download/v' + version + '/' + asset;
    const label = isWin ? 'Download for Windows' : 'Download for macOS';
    for (const node of btn.childNodes) {
      if (node.nodeType === Node.TEXT_NODE && node.textContent.trim()) {
        node.textContent = label + ' · v' + version;
        break;
      }
    }
  } catch { /* leave the fallback link as-is */ }
})();
`}</script>
